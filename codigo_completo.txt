import React, { useEffect, useState } from 'react';
import { FeedbackStatus } from '../types';

interface ProgressBarProps {
    total: number;
    current: number;
    status: FeedbackStatus;
}

export const ProgressBar: React.FC<ProgressBarProps> = ({ total, current, status }) => {
    const [flashColor, setFlashColor] = useState<'none' | 'green' | 'red'>('none');

    // Calculate percentage
    // Ensure it doesn't exceed 100% or drop below 0%
    const percentage = Math.min(100, Math.max(0, total > 0 ? (current / total) * 100 : 0));

    useEffect(() => {
        if (status === 'correct') {
            setFlashColor('green');
            const t = setTimeout(() => setFlashColor('none'), 500);
            return () => clearTimeout(t);
        } else if (status === 'incorrect' || status === 'timeout') {
            setFlashColor('red');
            const t = setTimeout(() => setFlashColor('none'), 500);
            return () => clearTimeout(t);
        } else {
            setFlashColor('none');
        }
    }, [status, current]); // Depend on current to trigger on progress change too if needed, but status is main trigger

    return (
        <div className="w-full max-w-md mx-auto mb-6">
            <div className="flex justify-between text-xs font-bold text-slate-400 uppercase tracking-wider mb-1">
                <span>Progress</span>
                <span>{current} / {total}</span>
            </div>
            <div className="h-4 bg-slate-700 rounded-full overflow-hidden relative shadow-inner">
                {/* Background Flash Layer */}
                <div
                    className={`absolute inset-0 transition-colors duration-300 ${flashColor === 'green' ? 'bg-green-500/20' :
                            flashColor === 'red' ? 'bg-red-500/20' : 'bg-transparent'
                        }`}
                />

                {/* Progress Fill */}
                <div
                    className={`h-full transition-all duration-500 ease-out relative ${flashColor === 'green' ? 'bg-green-500' :
                            flashColor === 'red' ? 'bg-red-500' :
                                'bg-gradient-to-r from-indigo-500 to-purple-500'
                        }`}
                    style={{ width: `${percentage}%` }}
                >
                    {/* Shine effect */}
                    <div className="absolute inset-0 bg-white/20" />

    );

import { normalizeAnswer } from './game';
export interface DiffChar { char: string; match: boolean; missing?: boolean }
export function alignAttempt(attempt: string, target: string): DiffChar[] {
  const a = Array.from(normalizeAnswer(attempt)), b = Array.from(normalizeAnswer(target));
  const d = Array.from({ length: a.length + 1 }, () => Array<number>(b.length + 1).fill(0));
  for (let i = 0; i <= a.length; i++) d[i][0] = i;
  for (let j = 0; j <= b.length; j++) d[0][j] = j;
  for (let i = 1; i <= a.length; i++) for (let j = 1; j <= b.length; j++)
    d[i][j] = Math.min(d[i-1][j] + 1, d[i][j-1] + 1, d[i-1][j-1] + Number(a[i-1] !== b[j-1]));
  const result: DiffChar[] = [];
  let i = a.length, j = b.length;
  while (i || j) {
    if (i && j && d[i][j] === d[i-1][j-1] + Number(a[i-1] !== b[j-1])) {
      result.push({ char: a[i-1], match: a[i-1] === b[j-1] }); i--; j--;
    } else if (j && d[i][j] === d[i][j-1] + 1) {
      result.push({ char: 'ï¼¿', match: false, missing: true }); j--;
    } else { result.push({ char: a[i-1], match: false }); i--; }
  }
  return result.reverse();
}

export const initialGame: GameState = {
  view: 'setup', allWords: [], bag: [], history: [], currentWord: '',
  streak: 0, maxStreak: 0, customDefs: {}, isLuckySession: false,
  feedback: 'idle', answer: '', hintUsed: false, turn: 0,

export const normalizeAnswer = (word: string) => word.normalize('NFC').trim().replace(/[â€™â€˜]/g, "'").replace(/\s+/g, ' ').toLowerCase();
export function parseWords(raw: string): string[] {
  const words = raw.split(/[\n,]+/).map(w => w.normalize('NFC').trim().replace(/\s+/g, ' ')).filter(Boolean);
  const invalid = words.find(w => !/^[\p{L}\p{M}]+(?:[ '\u2019\u2018-][\p{L}\p{M}]+)*$/u.test(w));
  if (invalid) throw new Error(`Please check this entry: â€œ${invalid}â€. Use letters, spaces, apostrophes or hyphens; separate entries with commas or new lines.`);
  return words;
}
export const mistakesFrom = (history: WordHistory[]) => [...new Set(history.filter(h => !h.correct || h.status === 'assisted').map(h => h.word))];
export const canResume = (s: GameState) => s.view !== 'summary' && !!(s.currentWord || s.bag.length);

export type GameAction =
  | { type: 'start'; words: string[]; lucky: boolean }
  | { type: 'restore'; state: GameState }
  | { type: 'resume' | 'exit' | 'finish' | 'hint' }
  | { type: 'answer'; answer: string; timeout: boolean; timestamp: number; turn: number }
  | { type: 'next'; loop: boolean; turn: number }
  | { type: 'definition'; word: string; definition: string };

export function gameReducer(s: GameState, action: GameAction): GameState {
  switch (action.type) {
    case 'restore': return action.state;
    case 'start': {
      if (!action.words.length) return s;
      return { ...initialGame, customDefs: s.customDefs, view: 'game', allWords: [...action.words],
        currentWord: action.words[0], bag: action.words.slice(1), isLuckySession: action.lucky, turn: s.turn + 1 };
    }
    case 'resume': return canResume(s) ? { ...s, view: 'game' } : s;
    case 'exit': return { ...s, view: 'setup' };
    case 'finish': return { ...s, view: 'summary', currentWord: '', bag: [] };
    case 'definition': return { ...s, customDefs: { ...s.customDefs, [action.word]: action.definition } };
    case 'hint': return s.feedback === 'idle' ? { ...s, hintUsed: true, streak: 0 } : s;
    case 'answer': {
      if (s.view !== 'game' || !s.currentWord || s.feedback !== 'idle' || action.turn !== s.turn) return s;
      if (!action.timeout && !action.answer.trim()) return s;
      const correct = !action.timeout && normalizeAnswer(action.answer) === normalizeAnswer(s.currentWord);
      const status: FeedbackStatus = action.timeout ? 'timeout' : correct ? (s.hintUsed ? 'assisted' : 'correct') : 'incorrect';
      const bag = [...s.bag];
      if (status !== 'correct') bag.splice(Math.min(bag.length, 3), 0, s.currentWord);
      const streak = status === 'correct' ? s.streak + 1 : 0;
      return { ...s, bag, streak, maxStreak: Math.max(streak, s.maxStreak), feedback: status, answer: action.answer,
        history: [...s.history, { word: s.currentWord, correct, status, timestamp: action.timestamp }] };
    }
    case 'next': {
      if (s.feedback === 'idle' || action.turn !== s.turn) return s;
      const bag = s.bag.length ? [...s.bag] : action.loop ? [...s.allWords] : [];
      if (!bag.length) return { ...s, view: 'summary', currentWord: '', bag: [] };
      return { ...s, currentWord: bag[0], bag: bag.slice(1), feedback: 'idle', answer: '', hintUsed: false, turn: s.turn + 1 };
    }
  }
}

// Saved v1 sessions used a queue with the active word prepended. v2 saves exact turn state.
export function restoreGame(value: unknown): GameState {
  if (!value || typeof value !== 'object') return initialGame;
  const p = value as Partial<GameState>;
  const validWords = (v: unknown): v is string[] => Array.isArray(v) && v.every(w => typeof w === 'string' && !!w.trim());
  if (!validWords(p.allWords) || !validWords(p.bag) || p.view === 'summary') return initialGame;
  const bag = [...p.bag];
  const currentWord = typeof p.currentWord === 'string' ? p.currentWord : (bag.shift() || '');
  const statuses = ['idle', 'correct', 'assisted', 'incorrect', 'timeout'];
  const feedback = p.feedback && statuses.includes(p.feedback) ? p.feedback : 'idle';
  return { ...initialGame, allWords: p.allWords, bag, currentWord, feedback,
    answer: typeof p.answer === 'string' ? p.answer : '', hintUsed: p.hintUsed === true,
    history: Array.isArray(p.history) ? p.history.filter(h => h && typeof h.word === 'string' && h.word && typeof h.correct === 'boolean' && Number.isFinite(h.timestamp)) : [],
    streak: Number.isFinite(p.streak) ? p.streak! : 0, maxStreak: Number.isFinite(p.maxStreak) ? p.maxStreak! : 0,
    turn: Number.isFinite(p.turn) ? p.turn! : 0 };
}

import { useState, useEffect } from 'react';

type Theme = 'dark' | 'light';

export function useTheme() {
    const [theme, setTheme] = useState<Theme>(() => {
        return document.documentElement.classList.contains('dark') ? 'dark' : 'light';
    });

    useEffect(() => {
        const root = window.document.documentElement;
        if (theme === 'dark') {
            root.classList.add('dark');
        } else {
            root.classList.remove('dark');
        }
        try { localStorage.setItem('theme', theme); } catch { /* The theme still works in memory. */ }
    }, [theme]);

    const toggleTheme = () => {
        setTheme((prev) => (prev === 'dark' ? 'light' : 'dark'));
    };

    return { theme, toggleTheme };
}

export class AudioService {
  private static ctx: AudioContext | null = null;
  private static audioCache = new Map<string, string>();
  private static pending = new Map<string, Promise<string>>();
  private static currentAudio: HTMLAudioElement | null = null;
  private static currentSpeakId = 0;
  private static finishSpeech: (() => void) | null = null;
  private static cacheCredential = '';

  private static getContext(): AudioContext {
    if (!this.ctx) {
      this.ctx = new (window.AudioContext || (window as any).webkitAudioContext)();
    }
    return this.ctx;
  }

  static resume() {
    try { const ctx = this.getContext(); if (ctx.state === 'suspended') void ctx.resume().catch(() => {}); }
    catch { /* Speech and spelling remain usable when Web Audio is unavailable. */ }
  }

  static stop() {
    this.currentSpeakId++;
    this.currentAudio?.pause();
    this.currentAudio = null;
    this.finishSpeech?.();
    this.finishSpeech = null;
    window.speechSynthesis?.cancel();
  }

  static playEffect(type: 'correct' | 'wrong') {
    if (!window.AudioContext && !(window as any).webkitAudioContext) return;
    this.resume();
    const ctx = this.getContext();
    const osc = ctx.createOscillator();
    const gain = ctx.createGain();

    osc.connect(gain);
    gain.connect(ctx.destination);

    const t = ctx.currentTime;

    if (type === 'correct') {
      osc.type = 'sine';
      osc.frequency.setValueAtTime(523, t); // C5
      osc.frequency.exponentialRampToValueAtTime(1046, t + 0.1); // C6
      gain.gain.setValueAtTime(0.1, t);
      gain.gain.linearRampToValueAtTime(0, t + 0.4);
      osc.start(t);
      osc.stop(t + 0.4);
    } else {
      osc.type = 'triangle';
      osc.frequency.setValueAtTime(150, t);
      osc.frequency.linearRampToValueAtTime(100, t + 0.2);
      gain.gain.setValueAtTime(0.1, t);
      gain.gain.linearRampToValueAtTime(0, t + 0.3);
      osc.start(t);
      osc.stop(t + 0.3);
    }
  }

  static async speak(text: string, voiceId: string, apiKey: string, rate = 1, language = 'en-US'): Promise<string | undefined> {
    this.stop();
    const myId = this.currentSpeakId;
    const playBrowser = async (): Promise<string | undefined> => {
      if (myId !== this.currentSpeakId) return;
      if (!window.speechSynthesis) return 'Speech is unavailable in this browser.';
      return new Promise(resolve => {
        const utterance = new SpeechSynthesisUtterance(text);
        utterance.lang = language;
        utterance.rate = rate;
        const finish = (message?: string) => {
          clearTimeout(watchdog);
          if (this.finishSpeech === cancel) this.finishSpeech = null;
          resolve(message);
        };
        const cancel = () => finish();
        const watchdog = window.setTimeout(() => {
          finish('Speech took too long. Press Listen to try again.');
          if (myId === this.currentSpeakId) window.speechSynthesis.cancel();
        }, 30000);
        this.finishSpeech = cancel;
        utterance.onend = () => finish();
        utterance.onerror = () => finish('Browser speech could not play. Check installed voices and audio settings.');
        window.speechSynthesis.speak(utterance);
      });
    };
    if (!apiKey || voiceId === 'browser') return playBrowser();
    // Rate changes reuse the same recording. Never put credentials in a cache key.
    if (apiKey !== this.cacheCredential) {
      this.audioCache.forEach(url => URL.revokeObjectURL(url));
      this.audioCache.clear();
      this.pending.clear();
      this.cacheCredential = apiKey;
    }
    const key = JSON.stringify([text, voiceId, language]);
    try {
      let url = this.audioCache.get(key);
      if (!url) {
        let pending = this.pending.get(key);
        if (!pending) {
          pending = (async () => {
            const response = await fetch(`https://api.elevenlabs.io/v1/text-to-speech/${encodeURIComponent(voiceId)}`, {
              method: 'POST', signal: AbortSignal.timeout(15000),
              headers: { 'xi-api-key': apiKey, 'Content-Type': 'application/json' },
              body: JSON.stringify({ text, model_id: 'eleven_turbo_v2_5', language_code: language.split('-')[0] }),
            });
            if (!response.ok) throw new Error('Speech request failed');
            const blob = await response.blob();
            if (apiKey !== this.cacheCredential) throw new Error('Voice settings changed');
            const objectUrl = URL.createObjectURL(blob);
            this.audioCache.set(key, objectUrl);
            if (this.audioCache.size > 100) {
              const oldest = this.audioCache.keys().next().value!;
              URL.revokeObjectURL(this.audioCache.get(oldest)!);
              this.audioCache.delete(oldest);
            }
            return objectUrl;
          })();
          this.pending.set(key, pending);
        }
        try { url = await pending; }
        finally { if (this.pending.get(key) === pending) this.pending.delete(key); }
      }
      if (myId !== this.currentSpeakId) return;
      const audio = new Audio(url);
      this.currentAudio = audio;
      audio.playbackRate = rate;
      await audio.play();
      return;
    } catch {
      if (myId !== this.currentSpeakId) return;
      const message = await playBrowser();
      return message || 'ElevenLabs could not play audio. Using your browser voice; check your key, quota or connection.';
    }
  }

const dictCache: Record<string, DictionaryEntry | null> = {};

export const fetchDefinition = async (word: string): Promise<DictionaryEntry | null> => {
  if (dictCache[word] !== undefined) return dictCache[word];

  try {
    const res = await fetch(`https://api.dictionaryapi.dev/api/v2/entries/en/${encodeURIComponent(word)}`, { signal: AbortSignal.timeout(10000) });
    if (!res.ok) throw new Error('Not found');
    const data = await res.json();

    if (Array.isArray(data) && data.length > 0) {
      dictCache[word] = data[0];
      return data[0];
    }
  } catch (e) {
    console.error("Dictionary fetch failed:", e);
    // silent fail
  }

  dictCache[word] = null;
  return null;
};

import React from 'react';
import { SetupView } from './components/SetupView';
import { GameView } from './components/GameView';
import { SummaryView } from './components/SummaryView';
import { useGameSession } from './hooks/useGameSession';

import { ThemeToggle } from './components/ThemeToggle';

export default function App() {
  const {
    config,
    setConfig,
    gameState,
    recentSessions,
    lastMistakes,
    hydrated,
    hasSavedState,
    storageError,
    dispatch,
    actions
  } = useGameSession();

  if (!hydrated) return null;

  return (
    <div className="min-h-screen flex items-center justify-center p-4 font-sans bg-gray-50 dark:bg-gray-900 transition-colors duration-300">
      <ThemeToggle />
      {storageError && <p role="alert" className="fixed bottom-2 left-2 right-2 z-50 rounded bg-red-100 p-3 text-red-800">{storageError}</p>}
      <div className="w-full max-w-xl">
        {gameState.view === 'setup' && (
          <SetupView
            config={config}
            setConfig={setConfig}
            onStart={actions.startSession}
            onResume={actions.resumeSession}
            hasSavedState={hasSavedState}
            recentSessions={recentSessions}
            lastMistakes={lastMistakes}
          />
        )}

        {gameState.view === 'game' && (
          <GameView
            gameState={gameState}
            config={config}
            dispatch={dispatch}
            onSaveDef={actions.saveCustomDef}
            onFinish={actions.handleSessionEnd}
            onExit={actions.exit}
          />
        )}

        {gameState.view === 'summary' && (
          <SummaryView
            history={gameState.history}
            maxStreak={gameState.maxStreak}
            allWords={gameState.allWords}
            lastMistakes={lastMistakes}
            onRestartSession={(words) => actions.startSession(words, !!gameState.isLuckySession)}
            onGoHome={actions.resetGame}
          />
        )}
      </div>
    </div>
  );
}
